1) write a shell script to input basic salary then generate salary slip 
------------------------------------------------------------------------------------------
echo "Enter Basic Salary:"
read basic

hra=$((basic * 20 / 100))
da=$((basic * 10 / 100))
gross=$((basic + hra + da))
pf=$((basic * 12 / 100))

net=$((gross - pf))

# Display salary slip
echo "================================="
echo "          SALARY SLIP"
echo "================================="
echo "Basic Salary       : Rs. $basic"
echo "HRA (20%)          : Rs. $hra"
echo "DA (10%)           : Rs. $da"
echo "---------------------------------"
echo "Gross Salary       : Rs. $gross"
echo "PF Deduction (12%) : Rs. $pf"
echo "---------------------------------"
echo "Net Salary         : Rs. $net"
echo "================================="

3) write a shell script to generate grade and percantage of students marks
-----------------------------------------------------------------------------------------------------------------------
echo "Enter OS mark:"
read OS
echo "Enter Python Mark:"
read Python
echo "Enter DBMS Mark:"
read DBMS

total=$((OS+Python+DBMS))
per=$((total/3))

echo "Total Marks : $total"
echo "Percentage : $per"

if [ $per -ge 90 ] && [ $per -le 100 ]; then
    echo "Grade A"
elif [ $per -ge 80 ] && [ $per -le 90 ]; then
    echo "Grade B"
elif [ $per -ge 70 ] && [ $per -le 80 ]; then
    echo "Grade C"
elif [ $per -ge 60 ] && [ $per -le 70 ]; then
    echo "Grade D"
else 
    echo "Fail"
fi
